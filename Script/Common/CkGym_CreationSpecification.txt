//============================================================================
// CK FRAMEWORK GYM CREATION SPECIFICATION
//============================================================================

/*
CRITICAL: This specification describes the gym FRAMEWORK — the project-agnostic
base classes, station spawning, auto-mode infrastructure, and cycler/HUD that
live in CkTests/Script/Common/. The framework itself has zero knowledge of
any specific gym.

CONSUMERS register their gyms at runtime from a project-level base game mode:
  - CkTests gyms: ACkTests_Gym_Base_GameMode → CkTests_Gyms::RegisterAll()
  - Your project:  AMyProject_Gym_Base_GameMode → MyProject_Gyms::RegisterAll()
Each level is self-contained — the Tab menu shows only gyms registered by that
level's game mode. See "PROJECT-LEVEL GAME MODES" below.
*/

//============================================================================
// 1. ARCHITECTURE OVERVIEW
//============================================================================

// Client-Server Safe Architecture:
// - All gym logic is in PlayerController (exists on both client/server)
// - No dependencies on GameMode availability timing
// - GameMode is minimal - just inheritance setup
// - PlayerController contains all metadata and startup logic

//============================================================================
// 2. BASE CLASSES (REQUIRED)
//============================================================================

// FRAMEWORK base classes (do not inherit directly — use project base):
//   ACk_Gym_Base_GameMode          — sets HUD, default pawn, station spawning
//   ACk_Gym_Base_PlayerController  — station lookup, grid layout, gym cycling
//   ACk_Gym_Base_Pawn              — ECS-ready DefaultPawn (use directly or subclass)
//
// PROJECT-LEVEL base game modes (inherit your gym GameMode from one of these):
//   CkTests:      ACkTests_Gym_Base_GameMode  (Script/Common/CkTests_Gym_Base_GameMode.as)
//   Your project: Create your own following the same pattern — a subclass of
//                 ACk_Gym_Base_GameMode that calls your registration namespace
//                 in BeginPlay. See CkTests_Gym_Base_GameMode.as as the template.
//
// These call their project's registration namespace in BeginPlay, which feeds
// the Tab menu via CkGym_Cycler::RegisterProjectGym().
//
// REQUIRED: A project-level base GameMode that overrides BeginPlay() MUST call
// Super::BeginPlay() AFTER its RegisterAll() call. The framework's
// ACk_Gym_Base_GameMode::BeginPlay() runs the startup-gym auto-travel hook
// (Editor Preferences -> Ck Tests -> Gym Cycler). Without the Super call,
// Default-mode and Last-mode auto-travel silently never fires — every PIE
// session lands on the launcher level regardless of the user's preference.
// The Super call must come AFTER RegisterAll() so the resolve runs against
// a populated registry.
//
// Canonical shape:
//   class AMyProject_Gym_Base_GameMode : ACk_Gym_Base_GameMode
//   {
//       UFUNCTION(BlueprintOverride)
//       void BeginPlay()
//       {
//           MyProject_Gyms::RegisterAll();
//           Super::BeginPlay();  // <-- runs the startup-gym auto-travel
//       }
//   }

        ck::Trace("✅ [Feature2] started at station location");

    // CONSOLE COMMANDS FOR TESTING
    //------------------------------------------------------------------------

    UFUNCTION(Exec, DisplayName="[GymName] Gym - Restart [Feature1]")
    void Ck_Gym[GymName]_Restart[Feature1]()
    {
        Request_Start[Feature1]();
    }

    UFUNCTION(Exec, DisplayName="[GymName] Gym - Restart [Feature2]")
    void Ck_Gym[GymName]_Restart[Feature2]()
    {
        Request_Start[Feature2]();
    }

    // Add more console commands as needed for individual feature testing

    //------------------------------------------------------------------------
    // CONTROL PANEL  (Script/Common/CkGym_ControlPanel.as)
    //
    // Put those commands ON SCREEN. A console command nobody can see is a
    // control nobody will use: the panel lists each one with the key that
    // fires it and its live value, and ACkGym_ControlPanelHUD - the base
    // GameMode's default HUDClass - does the drawing and the key polling.
    //
    // Rows are rebuilt every frame, so the value column reports live state.
    // Return no rows and no panel is drawn.
    //
    // Reserved keys a row must NOT use: Tab (gym cycler) and H (hide panel).
    // Row builders: Header / Status / Action / Toggle / ToggleNamed / Cycle /
    // Choice / Numbered (1-9 then 0, number row and numpad both).
    //------------------------------------------------------------------------

    FString Get_ControlPanelTitle() override
    {
        return "[GYMNAME]";
    }

    TArray<FCkGym_ControlRow> Get_ControlRows() override
    {
        auto Rows = TArray<FCkGym_ControlRow>();

        Rows.Add(CkGym_Control::Header("RE-RUN A STATION"));
        Rows.Add(CkGym_Control::Numbered(0, "[Feature1]", false));
        Rows.Add(CkGym_Control::Numbered(1, "[Feature2]", false));

        // Toggle / ToggleNamed report a value, so pass the REAL state - read it
        // back from whatever owns it. Mirror it in a member only when there is
        // no readback at all, and say so in a comment when you do.
        Rows.Add(CkGym_Control::Toggle(EKeys::E, "E", "[SomeToggle]", Get_[SomeToggle]()));

        // A control that needs an argument typed at it stays console-only. Say
        // so, rather than leaving a reader to conclude the gym cannot do it.
        Rows.Add(CkGym_Control::Status("[Something] with another value: console only"));

        return Rows;
    }

    // InRowIndex indexes the array Get_ControlRows() returned this frame.
    // Header and Status rows never arrive here, but they DO occupy an index -
    // keep conditional rows LAST so they cannot shift a keyed row's index.
    void Request_ControlActivated(int32 InRowIndex) override
    {
        if (InRowIndex == 1) { Ck_Gym[GymName]_Restart[Feature1](); }
        else if (InRowIndex == 2) { Ck_Gym[GymName]_Restart[Feature2](); }
        else if (InRowIndex == 3) { Request_Toggle[SomeToggle](); }
    }
}

// 8. STATION SETUP (AUTOMATIC)
//============================================================================

// Stations are now spawned automatically via Get_RequiredStations() override
// Base class handles:
// - Default grid layout (500 units forward, spaced 800 units along Y axis)
// - Stations face player (180° rotation)
// - Centered grid arrangement
// - Skip spawning if station already exists in level

// Station Spawn Parameters:
// - Tags: FName array for station identification
// - Title: FText displayed at top of station
// - Description: TArray<FText> for multi-paragraph descriptions
// - Transform: Optional override (defaults to grid layout)
// - AutoSize: Set true to size Width/Height from content (recommended)
// - Dimensions, colors, text scaling (use defaults unless needed)
//
// STATION SIZING — AUTO-SIZE FROM CONTENT:
// Set Station.AutoSize = true so the spawned UCk_EntityScript_GymStation
// computes Width/Height from the title + description text at spawn time
// (Apply_AutoSize_FromSpawnParams) and grows the alcove at runtime if
// Update_StationDisplay pushes longer text (Refit_FromMeasuredBounds).
// Avoids hardcoded dimensions that drift out of sync with content.
//
//   Station.AutoSize = true;
//
// Don't hand-tune Width/Height/Depth unless you specifically need a
// non-content-driven layout (e.g. the Station Showcase gym deliberately
// uses fixed dimensions to validate static-size paths). For any gym whose
// station's job is to display its title + description, AutoSize is the
// right answer.

// Manual Override:
// - Place BP_DemoDisplay in level with matching tag to override automatic spawn
// - Useful for custom positioning or special station configurations

// Tag Naming Rules:
// 1. Always start with "Gym."
// 2. Second level: broad category (Audio, Physics, Rendering, etc.)
// 3. Third level: specific feature being tested
// 4. Use PascalCase for readability
// 5. Keep names descriptive but concise

This allows gyms to:
- Display real-time statistics and counters
- Show attribute values and progress bars
- Update command lists dynamically
- Display network role information
- Show test results and status updates

Note: The FCkGym_Station_TitleAndDescription fragment is automatically handled by

LINE WRAPPING: The BP_DemoDisplay widget does not auto-wrap text. Any line longer
than roughly 55-60 characters will trail off the right edge of the station and

keep each line within that budget. This applies to AutoConfig.Description, any
f-string interpolation going into Fragment.Description, and any other text you
push to the display. Example:

    AutoConfig.Description =
        "First line, under ~55 chars.\n" +
        "Second line continues the thought.\n" +
        "Third line wraps up.";

If you find yourself needing many lines, consider trimming wording instead —
most existing gyms use 1-3 short lines total.
*/

//============================================================================
// 12. AUTO MODE PATTERN
//============================================================================

// GOAL: A gym should be immediately useful without any user input.
//
// When someone walks into a gym, they should:
//   1. See all stations actively running through their operations
//   2. See an [AUTO] indicator at the top of each station confirming it's live
//   3. See a step-by-step list of what the station is doing, with >> marking
//      the CURRENT operation — so the user understands the sequence and can
//      anticipate what comes next
//   4. See a description of what the station tests (persists after startup)
//   5. See a list of console commands they can type to interact manually
//   6. Know that typing any manual command will take them off auto mode for
//      that station, so they can explore at their own pace
//
// Not every station needs auto mode, but it's the default expectation.
// The biggest value is showing what the station is doing and what comes
// next, so the user builds a mental model of the feature without reading
// any code.
//
// DISPLAY LAYOUT (rendered by gym_auto:: helpers):
//
//   [AUTO] Running                        <- status (or [MANUAL])
//   Description of what this station      <- from AutoConfig.Description
//   tests, 1-2 lines.
//
//   Items: 5 (no limit)                   <- station-specific live data
//   [0] Potion [3/10]                        (counts, contents, grid, etc.)
//   ...
//
//   ===== Auto Sequence =====             <- plain english step descriptions
//      Add 3 potions                         >> marks the current step
//   >> Add 5 arrows                          user sees what's happening now
//      Sort inventory                        and what's coming next
//      Remove first (x3)
//
//   ===== Commands =====                  <- full console command names
//   Ck_GymMyFeature_AddPotion [n]            for manual interaction
//   Ck_GymMyFeature_SortAll
//   Ck_GymMyFeature_RestartAll
//
//   Ck_GymMyFeature_Auto [0/1]             <- global auto on/off
//   Ck_GymMyFeature_AutoUnbounded          <- per-station auto command
//
// IMPLEMENTATION:
//
// A station's step sequence is a CkStateMachine graph: ONE STATE CLASS PER
// STEP, transitions gated by a dwell condition. Helpers live in
// gym_sm:: (CkGym_StationSm.as).
//
// WHY NOT AN AutoStep COUNTER. The superseded shape held the real control flow
// in `AutoStep % TotalSteps` + an if-else chain, while FCkGym_AutoStep carried
// Description/FirstIndex/LastIndex purely for the HUD. The two were free to
// drift, and it is the exact anti-pattern CkStateMachine exists to remove
// ("Don't encode state transition logic in raw if-else chains" —
// CkStateMachine/Claude.md). Under an SM the HUD highlights the LIVE current
// state, so what is displayed is what is running, and the SM debugger's graph
// walk can inspect a station like any other state machine.
//
// STEP STATES (one UCLASS per step, usually a sibling _Steps.as file):
//     UCLASS()
//     class UCk_MyGym_Step_AddPotions : UCk_Gym_StepState
//     {
//         UFUNCTION(BlueprintOverride)
//         void DoDefineState(FCk_Handle_SmState_UnderConstruction& InHandle)
//         {
//             auto Trans = AddTransition(InHandle, UCk_MyGym_Step_Sort);
//             AddCondition(Trans, UCk_Gym_Dwell);      // _Short / _Long also available
//         }
//         UFUNCTION(BlueprintOverride)
//         void DoEnterState(FCk_Handle_SmState InHandle, ECk_Sm_NetContext InNetContext)
//         {
//             auto Station = Get_StationEntity();      // the SM's context owner
//             // ... act on the station ENTITY through utils_* ...
//         }
//     }
//
//     The last step transitions back to the first, closing the cycle.
//
//     A step state is its own entity script: it must NOT reach into the
//     station script's members. Read and write the station ENTITY via utils_*.
//     That is what lets each step be inspected independently, and it usually
//     deletes the station's parallel bookkeeping arrays outright.
//
// SETUP (in DoConstruct):
//     StepMachine = gym_sm::Setup(InHandle, UCk_MyGym_Step_AddPotions);
//
// CONFIGURATION (display only — the graph is the control flow):
//     StepConfig.Description = "What this station tests.";
//     StepConfig.GlobalAutoCommand = "Ck_GymMyFeature_Auto [0/1]";
//     StepConfig.PerStationAutoCommand = "Ck_GymMyFeature_AutoUnbounded";
//     StepConfig.Steps.Add(FCkGym_SmStep(UCk_MyGym_Step_AddPotions, "Add 3 potions"));
//     StepConfig.Steps.Add(FCkGym_SmStep(UCk_MyGym_Step_Sort,       "Sort inventory"));
//     StepConfig.ManualCommands.Add("Ck_GymMyFeature_AddPotion [n]");
//
// DISPLAY (in DisplayStats):
//     auto DisplayText = gym_sm::FormatHeader(StepConfig, StepMachine);
//     // ... station-specific live data ...
//     auto Trailer = gym_sm::FormatAutoAndCommands(StepConfig, StepMachine);
//     DisplayText = f"{DisplayText}{Trailer}";
//
// MANUAL HANDLERS (user took over — pause the sequence):
//     gym_sm::Request_SetAutoRunning(StepMachine, false);
//
// AUTO SET HANDLER — the Ck_Gym*_Auto console command still arrives as an
// FCk_Message_Gym_AutoSet broadcast (shared with stations not yet migrated):
//     UFUNCTION()
//     private void OnAutoSet(FCk_Handle InHandle, FGameplayTag InMessageName, FInstancedStruct InPayload)
//     {
//         auto Typed = InPayload.Get(FCk_Message_Gym_AutoSet);
//         gym_sm::Request_SetAutoRunning(StepMachine, Typed.Enabled);
//     }
//
// JUMP TO A STEP (reset commands, per-step manual commands):
//     gym_sm::Request_GoToStep(StepMachine, UCk_MyGym_Step_AddPotions);
//
// DWELL AND PAUSE. UCk_Gym_Dwell is an EventDriven condition that re-arms
// while the SM is paused. Do NOT reach for the framework's
// UCk_SmCondition_Timer here: it adds a plain CkTimer that ticks regardless of
// run status, so a paused station's dwell still elapses and the SM jumps the
// instant it is resumed — pause looks broken.
//
// PLAYER CONTROLLER COMMANDS:
//     Ck_Gym[Name]_Auto [0/1]         — forces ALL stations on or off
//     Ck_Gym[Name]_Auto[Station]      — re-enables a specific station
//     Use BroadcastAutoToAll/BroadcastAutoToTag helpers (see inventory gym PC).
//
// KEY DESIGN DECISIONS:
// - Use explicit on/off (not toggle) to avoid mixed state across stations
// - Auto starts ON by default so the gym is alive on entry
// - Manual commands auto-disable that station (user took over)
// - Step count must balance adds and removes — count carefully or items accumulate
// - Each auto step should perform the same operation a user could do manually
//
// MANUAL COMMAND DESIGN:
// Every station should have manual console commands that let the user
// interact directly with the feature being tested. The principle:
//
//   If the auto cycle does it, the user should be able to do it too.
//
// This means:
// 1. Each distinct operation in the auto cycle maps to a console command.
//    If auto does "add modifier" and "clear modifiers", expose both as
//    Ck_GymMyFeature_AddModifier and Ck_GymMyFeature_ClearModifiers.
//
// 2. Parameterized commands are valuable. If a station manipulates a
//    numeric value, accept it as a parameter with a sensible default:
//      UFUNCTION(Exec) void Ck_GymMyFeature_SetValue(float InValue = 100.0f)
//    This lets users experiment beyond what the auto cycle shows.
//
// 3. Every manual handler MUST call gym_auto::StopAuto() first:
//      gym_auto::StopAuto(AutoTimer, AutoRunning);
//    This switches the station to [MANUAL] so the user's action isn't
//    immediately overwritten by the next auto tick.
//
// 4. Commands are listed on the station display via AutoConfig.ManualCommands.
//    Include the parameter hint in the string:
//      AutoConfig.ManualCommands.Add("Ck_GymMyFeature_SetValue [val]");
//
// 5. Use dedicated USTRUCT messages for each command. Define them in the
//    gym's Common.as file. Parameterized commands carry data in the struct:
//      USTRUCT() struct FCk_Message_MyGym_SetValue { float32 Value; ... };
//    Parameterless commands use empty structs.
//
// 6. Entity scripts bind messages in DoBeginPlay and define handler UFUNCTIONs.
//    PlayerController defines UFUNCTION(Exec) commands that broadcast the
//    message to the appropriate entity tag.
//
// EXAMPLE — per-station command flow:
//
//   // Common.as — message definition
//   USTRUCT() struct FCk_Message_MyGym_SetValue { float32 Value; };
//
//   // Entity script — bind in DoBeginPlay
//   utils_messaging::BindTo_OnBroadcast(InHandle, FCk_Message_MyGym_SetValue,
//       FCk_Delegate_Messaging_OnBroadcast(this, n"OnSetValue"));
//
//   // Entity script — handler
//   UFUNCTION()
//   private void OnSetValue(FCk_Handle InHandle, FGameplayTag InMessageName, FInstancedStruct InPayload)
//   {
//       gym_auto::StopAuto(AutoTimer, AutoRunning);
//       auto Typed = InPayload.Get(FCk_Message_MyGym_SetValue);
//       // ... apply Typed.Value ...
//   }
//
//   // PlayerController — console command
//   UFUNCTION(Exec, DisplayName="My Gym - Set Value")
//   void Ck_GymMyFeature_SetValue(float InValue = 100.0f)
//   {
//       for (auto E : utils_entity_tag::ForEach_Entity(ck::ToEntity(this), n"TAG_MyStation"))
//       { utils_messaging::Broadcast(E, FCk_Message_MyGym_SetValue(InValue)); }
//   }
//
// STATION DESCRIPTION PERSISTENCE:
// The station's initial Description (set in Get_RequiredStations) is overwritten
// on the first DisplayStats tick. To keep it visible, set AutoConfig.Description
// which gym_auto::FormatHeader renders after the status line. Match the text in
// Get_RequiredStations so both the brief flash and the live display agree.
//
// IMPORTANT: Don't nest function calls with string args inside f-strings.
// f"{func("arg")}" breaks — the inner quotes terminate the f-string.
// Instead: DisplayText = DisplayText + func("arg");
//
// See CkGym_AutoStation.as for the full implementation.
// See CkInventoryGym_DataOnly_Unbounded.as for a reference station.

// instead of creating .uasset files in the content browser.
//
// Files containing asset blocks must use the _Assets.as suffix:
//   GOOD: CkInventoryGym_Assets.as
//   BAD:  CkInventoryGym_Shared.as (assets mixed with logic)

// PATTERN:
//   asset ItemDef_MyGym_Sword of UCk_InventoryItem_Definition
//   {
//       _CoreInfo = FCk_InventoryItem_CoreInfo(FText::FromString("Sword"));
//       auto Trait = Cast<UCk_ItemTrait_Tags>(NewObject(this, UCk_ItemTrait_Tags));
//       Trait._Tags = ...;
//       _ItemTraits.Add(Trait);
//   }
//
// ACCESSOR NAMESPACE:
//   namespace my_gym_items
//   {
//       UCk_InventoryItem_Definition Sword() { return ItemDef_MyGym_Sword; }
//   }
//
// The asset keyword creates a real UObject at module load — no .uasset on disk,

//============================================================================
// 14. REPLICATION GOTCHAS
//============================================================================

// The gym framework's PlayerController and station actors have Replicates=false.
// The Pawn has Replicates=true. This matters for any feature that routes through
// Request_BuildAndReplicate (e.g. item creation):
//
// - Item creation via Request_AddItemByDefinition requires the inventory's
//   ContextOwner to have replicated NetParams. In standalone this was fixed
//   by making Request_TryBuildAndReplicate build locally when the owner
//   doesn't replicate (see CkEntityReplicationDriver_Utils.cpp fix).
//
// - If you hit silent failures (e.g. Failed_NoSpaceAvailable on unbounded
//   inventories), check the replication chain: ContextOwner → NetParams →
//   Get_EntityReplication. The error name is misleading.
//
// - For multi-client PIE testing, ensure the entity creation path has a
//   replicated owner in the chain.

//============================================================================
// 15. LEVEL SETUP & BINARY PIPELINE (THINGS CLAUDE CAN'T SEE)
//============================================================================

// Several critical pieces of the gym framework are binary assets in the editor
// that AngelScript code depends on but cannot directly inspect:
//
// LEVEL REQUIREMENTS — Each gym level (.umap) must have:
//   1. World Settings → Default GameMode Override set to the project's base
//      game mode (e.g. ACkTests_Gym_Base_GameMode for CkTests, or your own
//      project base). Without this, no gyms register and the Tab menu is empty.
//   2. A Blueprint actor in the level that hooks onto the transient entity
//      and listens for messages that trigger BP_DemoDisplay spawning.
//      This is the bridge between the AS CkGym_Common::Request_SpawnNewStation()
//      call and the actual display hardware appearing in the world.
//
// BP_DemoDisplay PIPELINE — The display stations visible in-game are Blueprint
// actors (BP_DemoDisplay) with this internal flow:
//   1. BP_DemoDisplay spawns → it adds an entity script onto itself
//   2. That entity receives FCkGym_Station_TitleAndDescription dynamic fragments
//      (sent by the gym station scripts via Owner.AddOrGet_Fragment)
//   3. The fragment data (Title, Description) is rendered onto 3D text
//      components on the display actor each tick
//
// This means:
//   - Creating a new gym level: duplicate an existing gym level
//     (TestGyms_CkTests_Level or TestGyms_BB_MAP) so you get the
//     spawn-listener Blueprint for free.
//     Then change the World Settings game mode override.
//   - BP_DemoDisplay is a Content asset in Plugins/CkTests/Content/Common/DemoRoom/.
//     Both CkTests and BB levels reference it — it's part of the framework.
//   - If station text isn't updating, check that the entity script is writing to
//     the LIFETIME OWNER's fragment (utils_entity_lifetime::Get_LifetimeOwner),
//     not the station entity directly.
//
// HUD — ACkGym_MenuHUD (set by ACk_Gym_Base_GameMode via default HUDClass):
//   - Renders the Tab-key gym selector overlay
//   - Reads gym entries from CkGym_Cycler::Get_GymRegistry() (= Subsystem.ProjectGyms)
//   - Only shows gyms that were registered by the current level's game mode
//   - Info about which gyms exist comes through UCkGym_CyclerSubsystem
